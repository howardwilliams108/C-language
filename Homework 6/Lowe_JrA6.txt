// function definitions go here
void fillArray(ifstream &in, double arr[], int count)
{
    for (int i = 0; i < count; i++)
    {
        in >> arr[i];
    }
}

void printArray(double arr[], int count)
{
    int i;
    for(i = 0; i < count; i++)
    {
        cout << arr[i] << endl;
    }
}

void calculateAverage(double arr[], int size)
{
    double sum = 0.0;
    for(int i = 0; i < size; i++)
    {
        sum += arr[i];
    }
    double average = sum / size;
    cout.setf(ios::fixed);
    cout.setf(ios::showpoint);
    cout.precision(2);
    cout << "\nThe average of those values is: "<< average << endl;
}

void countValues(double arr[], int size)
{
    double value;
    int count = 0;
    
    cout << "Enter the value and I will count how many doubles in the array are greater than that value: ";
    cin >> value;
    
    // count how many are greater
    for(int i = 0; i < size; i++)
    {
        if (arr[i] > value)
        {
            cout << "The array value is: " << arr[i] << " index is " << i << endl;
            count++;
        }
    }
    if(count > 0)
    {
        // creating a second entry
        double *greater = new double[count];
        int great = 0;
        
        for(int i = 0; i < size; i++ )
        {
            if (arr[i] > value)
            {
                greater[great] = arr[i];
                great++;
                
            }
        }
        cout << "There are " << count << " in the array greater than " << value <<"." << endl;
        cout << "Here is a list of the values: "<< endl;
        
        printArray(greater, count);
        calculateAverage(greater, count);
        delete[] greater;
    }
}
