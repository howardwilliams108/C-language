//separates a phrase into words and stores each word in a string vector
void separatePhrase(const string& phrase, vector<string>& words)
{
    string currentWord; //hold the characters for the current word
    for (unsigned int i = 0; i < phrase.size(); i++)
    {
        if (phrase[i] != ' ')//not a space it will be a character in the word
        {
        //add the character to the current word
        currentWord += phrase[i];
        }
        else //a space has been found, the end of the word
        {
        //add the word to the vector use push_back
        words.push_back(currentWord);
        
        //clear the current word
        currentWord.clear();
        }
    }
    if (!currentWord.empty())//get the last word
    {
        //add the word to the vector use push_back
        words.push_back(currentWord);
    }
}
//input a vector of words, create a new phrase in reverse
string reversePhrase(vector<string> reverseWords)
{
    string reversed;
    while (!reverseWords.empty())
    {
        //get the last word
        reversed += reverseWords.back();
        
        //remove it from the vector
        reverseWords.pop_back();
        
        //add a space
        reversed += " ";
    }
    //return the phrase
    return reversed;
}
