// function definitions go here

//fills a vector of doubles from a file
//vectors are not automatically passed by reference 
void fillVector(ifstream &in, vector<double>& bitValues , int count)
{
    double value;
    
    for (int i = 0; i < count; i++)
    {
        in >> value;
        bitValues.push_back(value);
    }
}

//prints a vector of doubles onto the screen
// makes a copy of the vector pass by value
void printVector(vector<double> bitValues)
{
    int i;
    for(i = 0; i < bitValues.size(); i++)
    {
        cout << bitValues[i] << endl;
    }
}

//sorts a vector of doubles in ascending order
//make a copy of the vector pass by value
void sortVector(vector<double> bitValues)
{
    sort(bitValues.begin(), bitValues.end());
    cout << "\n\nprinting the sorted vector: ";
    printVector(bitValues);
}    

//calculates & prints the average of the elements in a vector
void calculateAverage(vector<double>greaterValues)
{
    double sum = 0.0;
    for(int i = 0; i < greaterValues.size(); i++)
    {
        sum += greaterValues[i];
    }
    double average = sum / greaterValues.size();
    cout.setf(ios::fixed);
    cout.setf(ios::showpoint);
    cout.precision(2);
    cout << "\nThe average of those values is: "<< average << endl;
}

//declare, ask and get a value

    cin >> value;

    for(int i = 0; i < bitValues.size(); i++)
    {

            greaterValues.push_back(bitValues[i]);

        cout << "There are " << greaterValues.size() << " in the array greater than " << value <<"." << endl;
        cout << "Here is a list of the values: "<< endl;

}
