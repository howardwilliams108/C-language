#include <string> // always included when using the string data type
#include <iostream> // the standard library for i/o
using namespace std; 

class Miniature
{
    public:
           Miniature(); // Miniature is the default constructor
           Miniature(string, string, string, int, double);
           
           //mutator functions
           void SetName(string);
           void SetOrigin(string);
           void SetLegion(string);
           void SetLid(int);
           void SetPrice(double);
           
           //accessor functions
           string GetName()const;
           string GetOrigin()const;
           string GetLegion()const;
           int GetLid()const;
           double GetPrice()const;
           
    
    private:
    
           string name, origin, legion;
           int lid;
           double price;
           
};

// Members functions follows below:

Miniature::Miniature() // this is the default constructor
{
    cout << "\ndefault Miniature is Roboute Guilliman";
    name = "Roboute Guilliman";
    origin = "Ultramar";
    legion = "Ultramarines";
    lid = 13;
    price = 73.50;
    
    
}

// explicit value  constructor follows below:
Miniature::Miniature(string n, string o, string le, int li, double p)
{
    cout << "\nExplicit value constructor";
    
    name = n;
    origin = o;
    legion = le;
    lid = li;
    price = p;
}

// five mutator functions will follow below:
void Miniature::SetName(string n) { name = n; } 
void Miniature::SetOrigin(string o) { origin = o; }
void Miniature::SetLegion(string le) { legion = le; }
void Miniature::SetLid(int li) { lid = li; }  
void Miniature::SetPrice(double p) { price = p; }

//five accessor functions will follow below:
string Miniature::GetName()const { return name; }
string Miniature::GetOrigin()const { return origin; }
string Miniature::GetLegion()const { return legion; }
int Miniature::GetLid()const { return lid; }
double Miniature::GetPrice()const { return price; }

//prints the info about a miniature onto the screen
void PrintMiniatureInfo(const Miniature& vtb)
{
    cout.setf(ios::fixed);
    cout.setf(ios::showpoint); // always used the code when necessary
    cout.precision(2);        // to set the number of decimal places for doubles
    
    cout << "\n\n$$$$$$$Printing the Information of the Miniatures$$$$$$\n\n";
    
    cout << "Miniature Name: " << vtb.GetName() << endl;
