Miniature findSelectedMini(Miniature miniList[], int count, int legion)
{
    for(int i = 0; i < count; i++)
    {
        if(miniList[i].getLegion() == legion)
        {
            cout << "found the mini: ";
            miniList[i].display();
            return miniList[i];
        }
    }
    cout << "\n$$$$ The miniature was not found! Returning the first miniature in the list $$$$\n";
    return miniList[0];
}

//load miniature data from the file
//connect to the file inside the Function
void loadMiniatures(Miniature miniList[], int& count)
{
    ifstream infile;
    infile.open("Miniatures.txt");
    int legion;
    string name, species;
    double cost, price;
    
    if(!infile) {
        cerr << "Error opening file" << endl;
        exit(1);
    }
    
     // While loop that checks if all variables can be successfully read
    while (!infile.eof()) {

        infile >> legion >> name >> species >> cost >> price;

        // Debugging output
        cout << "\nLoaded: " << legion << ", " << name << ", " << species << ", $" << cost << ", $" << price << endl;

        // Add the new Miniature object to the list
        miniList[count].setLegion(legion);
        miniList[count].setName(name);
        miniList[count].setSpecies(species);
        miniList[count].setCost(cost);
        miniList[count].setPrice(price);
        
        //add the rest of the values
        count++;

    }

    // Check for any errors or incomplete data
    if (infile.eof()) {
        cout << "\n\n*****Finished reading the file*****\n";
    }
    else {
        cerr << "Error reading data from the file.\n";
    }
    //close the file connection
    infile.close();
}

 // inputs the number of miniatures and returns the profit
double Miniature::calculateProfit(double minis) const
{
    return (priceperMini - costperBook) * minis;
}

// overload the subtraction(-) operator
// calculate the difference in price of two miniatures:
double operator-(const Miniature& m1, const Miniature& m2)
{
    return m1.priceperMini - m2.priceperMini;
}

//overload the less than (<) operator
//compare the price of two miniatures:
bool operator<(const Miniature& m1, const Miniature& m2)
{
    return m1.priceperMini < m2.priceperMini;
}
